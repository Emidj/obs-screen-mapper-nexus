cmake_minimum_required(VERSION 3.28)
project(screen_mapper_editor VERSION 1.0.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
add_library(mapper-editor SHARED editor.cpp preview.cpp presets.cpp ../src/version.rc)
target_compile_definitions(mapper-editor PRIVATE MAPPER_EDITOR)
target_include_directories(mapper-editor PUBLIC . "${CORE_BUILD}")
target_include_directories(mapper-editor SYSTEM PUBLIC "${OBS_HEADERS}/libobs")
target_compile_definitions(mapper-editor PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(mapper-editor PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8;/permissive->")
target_link_libraries(mapper-editor PRIVATE Qt6::Widgets "${OBS_IMPORT}" d2d1 d3d11 ole32 shlwapi xmllite)
set_target_properties(mapper-editor PROPERTIES OUTPUT_NAME obs-screen-mapper-ui)

target_include_directories(editor-test PRIVATE . "${CORE_BUILD}")
